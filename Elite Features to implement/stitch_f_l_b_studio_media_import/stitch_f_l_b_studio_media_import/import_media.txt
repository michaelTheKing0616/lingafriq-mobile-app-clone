function switchView(btn, id) {
  document.querySelectorAll('.vt-btn').forEach(b => b.classList.remove('active'));
  document.querySelectorAll('.view-panel').forEach(p => p.classList.remove('active'));
  btn.classList.add('active');
  document.getElementById('view-' + id).classList.add('active');
}

function selectMedia(el) {
  document.querySelectorAll('.media-item').forEach(i => i.classList.remove('active'));
  el.classList.add('active');
}

function setRpTab(btn, id) {
  document.querySelectorAll('.rp-tab').forEach(t => t.classList.remove('active'));
  btn.classList.add('active');
  document.querySelectorAll('#rp-vocab, #rp-progress, #rp-ai').forEach(p => p.style.display = 'none');
  document.getElementById('rp-' + id).style.display = 'block';
}

function showProcessing() {
  document.getElementById('processing-overlay').classList.add('show');
}

function closeProcessing() {
  document.getElementById('processing-overlay').classList.remove('show');
}

function handleQuiz(el, type) {
  document.querySelectorAll('.qc-option').forEach(o => o.classList.remove('correct','wrong'));
  el.classList.add(type);
  document.getElementById('quiz-explain').classList.add('show');
}
